```js
// bin/lib/pr-checks/classify.js — pure classifier for a run PR's
// statusCheckRollup at HEAD (#3013). The red/pending/empty/green rules are
// _shared/pr-first-merge.md Step 2.5's, applied in the same order (red
// first); that prose stays the single authority and this module is its code
// twin — tests/pr-check-gate-conformance.test.js pins FAILING_CONCLUSIONS
// against the prose list. Two additions the merge gate does not need:
//   - `stale`: the rollup's headRefOid is not local HEAD (the newest commit
//     is not pushed or not yet checked) — nothing is concluded from it.
//   - StatusContext entries ({context, state, targetUrl}) carry no
//     status/conclusion, so they are normalized into that shape first;
//     otherwise a literal reading of "status not COMPLETED → pending" would
//     hold a finished commit status pending forever.
// Any failing check is red: statusCheckRollup does not mark which checks are
// required, and the merge gate already treats every failing check as red.
'use strict';

const FAILING_CONCLUSIONS = Object.freeze(['FAILURE', 'TIMED_OUT', 'ERROR', 'STARTUP_FAILURE']);

function normalizeEntry(entry) {
  const e = entry && typeof entry === 'object' ? entry : {};
  if (typeof e.context === 'string' && !('status' in e)) {
    const state = String(e.state || '').toUpperCase();
    const done = state === 'SUCCESS' || state === 'FAILURE' || state === 'ERROR';
    return { name: e.context, url: e.targetUrl || null, status: done ? 'COMPLETED' : 'PENDING', conclusion: done ? state : null };
  }
  return { name: e.name || null, url: e.detailsUrl || null, status: e.status || null, conclusion: e.conclusion || null };
}

function result(state, headRefOid, extra = {}) {
  return { state, failing: [], pending: [], headRefOid: headRefOid || null, ...extra };
}

function classifyRollup({ rollup, headRefOid, localHead, hasPrCi } = {}) {
  if (!Array.isArray(rollup) || typeof headRefOid !== 'string' || !headRefOid) {
    return result('read-failed', headRefOid, { reason: 'malformed-rollup' });
  }
  if (typeof localHead !== 'string' || !localHead || headRefOid !== localHead) {
    return result('stale', headRefOid);
  }
  const entries = rollup.map(normalizeEntry);
  const failing = entries
    .filter((e) => FAILING_CONCLUSIONS.includes(e.conclusion))
    .map((e) => ({ name: e.name, url: e.url }));
  if (failing.length > 0) return result('red', headRefOid, { failing });
  const pending = entries.filter((e) => e.status !== 'COMPLETED').map((e) => e.name);
  if (pending.length > 0) return result('pending', headRefOid, { pending });
  if (entries.length === 0) return result(hasPrCi ? 'pending' : 'no-ci', headRefOid);
  return result('green', headRefOid);
}

module.exports = { FAILING_CONCLUSIONS, normalizeEntry, classifyRollup };
```

#!/usr/bin/env node
// bin/pr-checks.js -- read a pr-first run's own PR check rollup at HEAD and
// classify it (#3013). The one shared read /claude-tweaks:test's pipeline
// gate and /claude-tweaks:review Step 1.5 both run, per
// skills/_shared/pr-check-gate.md (which owns how each state is routed).
// Classification rules: _shared/pr-first-merge.md Step 2.5, via
// lib/pr-checks/classify.js; the merge gate keeps its own read.
//
//   node bin/pr-checks.js --run <run-dir> [--help]
//
// Prints one JSON line: {state, failing:[{name,url}], pending:[name],
// headRefOid, localHead, pr, reason?}. state is red | pending | stale |
// green | no-ci | read-failed | no-pr | gh-absent.
//
// Status is data: exit 0 in every state, including no-pr and read-failed.
// 2 malformed invocation; 3 run dir missing or not anchored under the main
// checkout (a worktree-local shadow -- _shared/pipeline-run-dir.md, [IL-127]).
'use strict';

const { execFileSync } = require('child_process');
const { classifyRollup } = require('./lib/pr-checks/classify');
const { resolveTarget } = require('./lib/log-decision/append');
const { readRunStateWithParent } = require('./lib/hooks/context');
const { ghAvailable } = require('./lib/repo-resolve');
const { hasPullRequestCi } = require('./lib/merge-verification');
const { GH_TIMEOUT_MS } = require('./lib/shared-primitives');

const USAGE = 'usage: pr-checks.js --run <run-dir> [--help]\n'
  + '  prints {state, failing, pending, headRefOid, localHead, pr, reason?};\n'
  + '  state: red | pending | stale | green | no-ci | read-failed | no-pr | gh-absent\n'
  + '  exit 0 in every state; 2 malformed invocation; 3 run dir missing or not anchored\n';

function parseArgs(argv) {
  const o = { run: null, help: false };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === '--help' || a === '-h') o.help = true;
    else if (a === '--run') o.run = argv[++i] ?? null;
    else return { error: `unknown argument: ${a}` };
  }
  return o;
}

const QUIET = { encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'], windowsHide: true };

const realDeps = {
  cwd: () => process.cwd(),
  mainRoot: undefined,
  ghAvailable: () => ghAvailable(),
  hasPullRequestCi: (root) => hasPullRequestCi(root),
  gh: (args) => execFileSync('gh', args, { ...QUIET, timeout: GH_TIMEOUT_MS }),
  git: (args) => execFileSync('git', args, QUIET),
  readRunState: (runDir) => readRunStateWithParent(runDir, { requireRunIdParent: true }).state,
  stdout: (s) => process.stdout.write(s),
  stderr: (s) => process.stderr.write(s),
};

function errText(e) {
  return [e && e.message, e && e.stderr, e && e.stdout].filter(Boolean).join(' ').trim() || String(e);
}

function report(deps, body) {
  deps.stdout(`${JSON.stringify({ failing: [], pending: [], headRefOid: null, localHead: null, pr: null, ...body })}\n`);
  return 0;
}

function run(argv, overrides = {}) {
  // Tests pass only the seams they fake; readRunState stays the real
  // run-state.json reader unless a test overrides it.
  const deps = { ...realDeps, ...overrides };
  const o = parseArgs(argv);
  if (o.error) { deps.stderr(`pr-checks.js: ${o.error}\n${USAGE}`); return 2; }
  if (o.help) { deps.stdout(USAGE); return 0; }
  if (!o.run) { deps.stderr(`pr-checks.js: --run <run-dir> is required\n${USAGE}`); return 2; }
  const target = resolveTarget({ runDir: o.run, cwd: deps.cwd(), mainRoot: deps.mainRoot });
  if (!target.ok) {
    if (target.reason === 'missing') deps.stderr(`pr-checks.js: run dir does not exist: ${o.run}\n`);
    else deps.stderr(`pr-checks.js: run dir is not anchored under the main checkout (a worktree-local shadow): ${o.run} — resolve $RUN_ROOT per _shared/pipeline-run-dir.md's Anchoring section and pass the main-checkout path\n`);
    return 3;
  }
  let state = null;
  try { state = deps.readRunState(o.run); } catch { state = null; }
  const n = state && state.pr && state.pr.number;
  if (!Number.isInteger(n) || n <= 0) return report(deps, { state: 'no-pr' });
  if (!deps.ghAvailable()) return report(deps, { state: 'gh-absent', pr: n });
  let localHead;
  try { localHead = String(deps.git(['rev-parse', 'HEAD'])).trim(); } catch (e) {
    return report(deps, { state: 'read-failed', pr: n, reason: `local HEAD unreadable: ${errText(e)}` });
  }
  let view;
  try { view = JSON.parse(deps.gh(['pr', 'view', String(n), '--json', 'headRefOid,statusCheckRollup'])); } catch (e) {
    return report(deps, { state: 'read-failed', pr: n, localHead, reason: `gh pr view failed: ${errText(e)}` });
  }
  let hasPrCi = false;
  try { hasPrCi = Boolean(deps.hasPullRequestCi(deps.cwd())); } catch { hasPrCi = false; }
  const verdict = classifyRollup({ rollup: view && view.statusCheckRollup, headRefOid: view && view.headRefOid, localHead, hasPrCi });
  return report(deps, { ...verdict, localHead, pr: n });
}

module.exports = { run, parseArgs };

if (require.main === module) process.exitCode = run(process.argv.slice(2), realDeps);
```

```markdown
# PR-Check Gate — the run PR's CI state at a test/review gate

Under `integration-model: pr-first` (`_shared/integration-model.md`) a run's draft PR exists from
run start and every phase exit pushes to it (`_shared/pr-early-run-lifecycle.md`), so CI on that
PR can already be red at the commit a local verification pass just blessed — a race that shows
only on the CI runner (#3013's incident: a hydration-timing e2e failure that never reproduced
locally). This procedure is the one read both `/claude-tweaks:test`'s pipeline gate and
`/claude-tweaks:review` Step 1.5 run so that red check stops the run before review, polish, and
wrap-up are spent on top of it. Callers cite this file; neither restates it.

**Classification authority.** The red/pending/empty/green rules are
`_shared/pr-first-merge.md`'s Step 2.5, applied in that order by `bin/lib/pr-checks/classify.js`
(any failing check is red — the rollup does not mark which checks are required, and the merge gate
already treats every failing check as red). This read adds only `stale` — the PR's `headRefOid` is
not local HEAD, so nothing is concluded. The merge gate keeps its own read and stays the final
authority; this gate never replaces it.

## When it runs

Only when `$PIPELINE_RUN_DIR` is set (a standalone run has no run PR to read — skip, nothing to
log), and only after the caller's own local verification has passed:

```bash
node "${CLAUDE_PLUGIN_ROOT}/bin/pr-checks.js" --run "$PIPELINE_RUN_DIR"
```

One plain command; it prints `{state, failing, pending, headRefOid, localHead, pr, reason?}` and
exits 0 in every state. Exit 2 or 3 is a tooling problem with the read itself — treat it exactly
like `read-failed` below.

## Routing by `state`

- **`red`** — a gate failure, the same class as a local test failure: do not set `TEST_PASSED`
  (`/test`), and **STOP** (`/review` Step 1.5). Render each `failing[]` entry's name and url —
  "PR #{pr} check `{name}` is failing at {headRefOid short} ({url})" — as the failure card's
  finding. A green local run is not evidence against it: fix forward, push, and re-run the gate.
- **`pending`** or **`stale`** — pass, with a caveat: `TEST_PASSED` is set and carries the caveat
  naming the pending checks (or the unpushed HEAD), and one entry is logged:
  `node "${CLAUDE_PLUGIN_ROOT}/bin/log-decision.js" --run "$PIPELINE_RUN_DIR" --status AUTO --section "{/test|/review}" --step "PR-check gate" --reversibility high --text "PR #{pr} checks {state} at {headRefOid short} ({pending names, or 'HEAD not yet pushed'}) — passed with caveat"`.
  The merge gate's bounded watch settles them later.
- **`green`** — pass; nothing to log.
- **`no-pr`**, **`gh-absent`**, **`no-ci`**, or **`read-failed`** (and exit 2/3) — fail open
  exactly as before this gate existed: the caller's local result stands. Write one `SKIP` entry
  per `_shared/auto-decision-log.md`'s degrade-trace rule:
  `node "${CLAUDE_PLUGIN_ROOT}/bin/log-decision.js" --run "$PIPELINE_RUN_DIR" --status SKIP --section "{/test|/review}" --step "PR-check gate" --reversibility n/a --text "condition: pr-checks.js {state}{: reason} → fallback: local verification only"`.
  `no-pr` is the `local-merge` (and pre-PR) case; `read-failed` is never read as green — it only
  means this gate could not tell, and the merge gate's own read will.
```
