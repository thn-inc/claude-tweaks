# QA Pass Stamp Implementation Plan (#3003)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A dispatched record's second pipeline call skips QA stories that already passed on the same clean commit, by reading a runner-written QA pass stamp bound to HEAD and to each story file's content.

**Architecture:** A pure lib module (`plugin/bin/lib/verify/qa-stamp.js`) composes, validates, reads, and writes `{git-dir}/claude-tweaks-qa-pass.json` and decides skip-vs-run. A thin CLI (`plugin/bin/qa-stamp.js`, `run(argv, deps)` seam like `plugin/bin/pr-checks.js`) wires it to git and the filesystem. `/claude-tweaks:test`'s skill prose calls the CLI: `--write` after Phase 5 writes `report.json`, `--check` before pipeline story dispatch. The pipeline bullets now route story runs through `qa-procedures.md` explicitly, so Phase 5 (and its `report.json`) always runs.

**Tech Stack:** Node 18+ CommonJS, `node:test`, `node:crypto` sha256, no dependencies.

**Spec:** `.claude-tweaks/pipelines/2026-10-16T172900-spec-3072-3003/spec-3003/work/3003-spec.md`

## Review Focus

1. A story YAML edited after the stamp was written (including Phase 4.5's own selector recoveries) — the stamp must not cover it. Pinned in Task 1 (`story-changed`).
2. A selected story whose id exists in the stamp but under a different file — must run. Pinned in Task 1 (`story-not-covered`).
3. A `report.json` whose `summary` says clean but whose `stories` array has a FAIL or SKIPPED entry (or vice versa) — must write no stamp. Pinned in Task 1.
4. A stamp written for a different `STORIES_DIR` than the current selection — must run. Pinned in Task 1 (`stories-dir-mismatch`).
5. A hand-edited or truncated stamp file (bad sha, missing `stories`, non-hex hash) — read as absent, never as a covering stamp. Pinned in Task 1 (`readQaStamp`).

```js
// plugin/bin/lib/verify/qa-stamp.js — the QA pass stamp (#3003). Records that a
// pipeline QA run passed every selected story at one clean commit, so a later
// /claude-tweaks:test call at the same HEAD (a dispatched group's second call)
// can skip re-running them. Bound to HEAD and to each story file's sha256: a
// story YAML edited since the stamp (Phase 4.5's selector recoveries included)
// is never covered. Lives beside the verification stamp in the checkout's own
// git dir (report.js's gitDir()), so both dispatch calls in one worktree read
// it. bin/qa-stamp.js is the only writer. Reads fail toward absence, like
// stamp.js: anything malformed is null, never a covering stamp.
'use strict';

const fs = require('fs');
const path = require('path');
const { writeFileAtomic } = require('../atomic-write');

const QA_STAMP_NAME = 'claude-tweaks-qa-pass.json';
const SHA_RE = /^[0-9a-f]{40}$/;
const HASH_RE = /^[0-9a-f]{64}$/;
const PASSED = new Set(['PASS', 'PASS_WITH_CAVEATS']);

function composeQaStamp({
  report, head, dirty, storyHashes, storiesDir, reportPath, at,
}) {
  const stories = report && Array.isArray(report.stories) ? report.stories : [];
  if (stories.length === 0) return { error: 'no-stories' };
  for (const s of stories) {
    if (!PASSED.has(s.status)) return { error: `story-not-passed: ${s.id} ${s.status}` };
  }
  const summary = (report && report.summary) || {};
  if ((summary.failed || 0) > 0 || (summary.skipped || 0) > 0) return { error: 'summary-not-clean' };
  if (summary.total !== stories.length) return { error: 'summary-mismatch' };
  if (dirty !== false) return { error: 'dirty-tree' };
  if (typeof head !== 'string' || !SHA_RE.test(head)) return { error: 'no-head' };
  const entries = [];
  for (const s of stories) {
    const sha256 = storyHashes[s.source_file];
    if (typeof sha256 !== 'string') return { error: `story-file-unreadable: ${s.source_file}` };
    entries.push({ id: s.id, file: s.source_file, sha256 });
  }
  return {
    stamp: {
      sha: head, dirty: false, stories: entries, storiesDir, at, reportPath,
    },
  };
}

function decideQaSkip({
  stamp, head, dirty, storiesDir, selected,
}) {
  if (!stamp) return { skip: false, reason: 'no-stamp' };
  if (dirty !== false) return { skip: false, reason: 'dirty-tree' };
  if (stamp.sha !== head) return { skip: false, reason: 'sha-mismatch' };
  if (stamp.storiesDir !== storiesDir) return { skip: false, reason: 'stories-dir-mismatch' };
  if (!Array.isArray(selected) || selected.length === 0) return { skip: false, reason: 'nothing-selected' };
  for (const sel of selected) {
    const entry = stamp.stories.find((s) => s.id === sel.id && s.file === sel.file);
    if (!entry) return { skip: false, reason: `story-not-covered: ${sel.id}` };
    if (entry.sha256 !== sel.sha256) return { skip: false, reason: `story-changed: ${sel.id}` };
  }
  return { skip: true, reason: 'covered' };
}

function validStory(s) {
  return s !== null && typeof s === 'object' && typeof s.id === 'string'
    && typeof s.file === 'string' && typeof s.sha256 === 'string' && HASH_RE.test(s.sha256);
}

function readQaStamp(gitDir, fsImpl = fs) {
  let parsed;
  try { parsed = JSON.parse(fsImpl.readFileSync(path.join(gitDir, QA_STAMP_NAME), 'utf8')); } catch { return null; }
  if (parsed === null || typeof parsed !== 'object') return null;
  if (typeof parsed.sha !== 'string' || !SHA_RE.test(parsed.sha)) return null;
  if (parsed.dirty !== false) return null;
  if (!Array.isArray(parsed.stories) || !parsed.stories.every(validStory)) return null;
  return parsed;
}

function writeQaStamp(gitDir, stamp, fsImpl = fs) {
  const p = path.join(gitDir, QA_STAMP_NAME);
  writeFileAtomic(p, `${JSON.stringify(stamp, null, 2)}\n`, {
    writeFile: fsImpl.writeFileSync,
    rename: fsImpl.renameSync,
    unlink: fsImpl.unlinkSync || (() => {}),
  });
  return p;
}

module.exports = {
  composeQaStamp, decideQaSkip, readQaStamp, writeQaStamp, QA_STAMP_NAME,
};
```

```js
#!/usr/bin/env node
// bin/qa-stamp.js — write and check the QA pass stamp (#3003), the only writer
// of {git-dir}/claude-tweaks-qa-pass.json (lib/verify/qa-stamp.js holds the
// rules). /claude-tweaks:test runs --write after QA Phase 5 writes report.json,
// and --check before pipeline story dispatch, so a dispatched group's second
// call skips stories that already passed at the same clean HEAD.
//
//   node bin/qa-stamp.js --write --report <RUN_DIR/report.json> [--stories-dir <dir>]
//   node bin/qa-stamp.js --check --stories <id=file,...> [--stories-dir <dir>]
//
// Prints one JSON line: --write → {written: true, path, sha, stories} or
// {written: false, reason}; --check → {skip, reason, sha}. Status is data:
// exit 0 in every state; 2 malformed invocation.
'use strict';

const fs = require('fs');
const path = require('path');
const crypto = require('crypto');
const {
  composeQaStamp, decideQaSkip, readQaStamp, writeQaStamp,
} = require('./lib/verify/qa-stamp');
const { gitInfo, gitDir } = require('./lib/verify/report');

const USAGE = 'usage: qa-stamp.js --write --report <RUN_DIR/report.json> [--stories-dir <dir>]\n'
  + '       qa-stamp.js --check --stories <id=file,...> [--stories-dir <dir>]\n'
  + '  prints one JSON line; exit 0 in every state; 2 malformed invocation\n';

function parseArgs(argv) {
  const o = {
    write: false, check: false, report: null, storiesDir: null, stories: null,
  };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === '--write') o.write = true;
    else if (a === '--check') o.check = true;
    else if (a === '--report') o.report = argv[++i] ?? null;
    else if (a === '--stories-dir') o.storiesDir = argv[++i] ?? null;
    else if (a === '--stories') {
      const raw = argv[++i] ?? '';
      const pairs = raw.split(',').filter(Boolean).map((item) => {
        const eq = item.indexOf('=');
        return eq > 0 && eq < item.length - 1 ? { id: item.slice(0, eq), file: item.slice(eq + 1) } : null;
      });
      if (pairs.length === 0 || pairs.includes(null)) return { error: `--stories expects id=file[,id=file...]: ${raw}` };
      o.stories = pairs;
    } else return { error: `unknown argument: ${a}` };
  }
  if (o.write === o.check) return { error: 'exactly one of --write or --check is required' };
  if (o.write && !o.report) return { error: '--write requires --report <path>' };
  if (o.check && !o.stories) return { error: '--check requires --stories <id=file,...>' };
  return o;
}

const realDeps = {
  cwd: () => process.cwd(),
  gitInfo: () => gitInfo(),
  gitDir: () => gitDir(),
  now: () => new Date().toISOString(),
  stdout: (s) => process.stdout.write(s),
  stderr: (s) => process.stderr.write(s),
};

function hashFile(p) {
  try { return crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex'); } catch { return null; }
}

function emit(deps, body) {
  deps.stdout(`${JSON.stringify(body)}\n`);
  return 0;
}

function write(o, deps) {
  let report;
  try { report = JSON.parse(fs.readFileSync(o.report, 'utf8')); } catch {
    return emit(deps, { written: false, reason: 'report-unreadable' });
  }
  const dir = deps.gitDir();
  if (!dir) return emit(deps, { written: false, reason: 'no-git-dir' });
  const storiesDir = o.storiesDir || report.stories_dir || 'stories';
  const storyHashes = {};
  for (const s of Array.isArray(report.stories) ? report.stories : []) {
    storyHashes[s.source_file] = hashFile(path.join(deps.cwd(), storiesDir, String(s.source_file)));
  }
  const { sha, dirty } = deps.gitInfo();
  const { stamp, error } = composeQaStamp({
    report, head: sha, dirty, storyHashes, storiesDir, reportPath: path.resolve(o.report), at: deps.now(),
  });
  if (error) return emit(deps, { written: false, reason: error });
  const p = writeQaStamp(dir, stamp);
  return emit(deps, { written: true, path: p, sha: stamp.sha, stories: stamp.stories.length });
}

function check(o, deps) {
  const dir = deps.gitDir();
  const stamp = dir ? readQaStamp(dir) : null;
  const storiesDir = o.storiesDir || 'stories';
  const selected = o.stories.map((s) => ({ ...s, sha256: hashFile(path.join(deps.cwd(), storiesDir, s.file)) }));
  const { sha, dirty } = deps.gitInfo();
  const verdict = decideQaSkip({
    stamp, head: sha, dirty, storiesDir, selected,
  });
  return emit(deps, { ...verdict, sha: stamp ? stamp.sha : null });
}

function run(argv, overrides = {}) {
  const deps = { ...realDeps, ...overrides };
  const o = parseArgs(argv);
  if (o.error) { deps.stderr(`qa-stamp.js: ${o.error}\n${USAGE}`); return 2; }
  return o.write ? write(o, deps) : check(o, deps);
}

module.exports = { run, parseArgs };

if (require.main === module) process.exitCode = run(process.argv.slice(2), realDeps);
```
